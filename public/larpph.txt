local function requestURL(url)
    local fn = request or http_request or (syn and syn.request)
    if not fn then
        error("No request function available")
    end

    local response = fn({
        Url = url,
        Method = "GET"
    })

    return response.Body
end

local function ensureFolder(path)
    if not isfolder(path) then
        makefolder(path)
    end
end

local function formatTime(seconds)
    seconds = math.max(0, math.floor(seconds))
    local minutes = math.floor(seconds / 60)
    local secs = seconds % 60
    return string.format("%d:%02d", minutes, secs)
end

local function applyEnhancement()
    Equalizer.Enabled = false
    Reverb.Enabled = false

    if Settings.Enhancement == "Bass Boost" then
        Equalizer.Enabled = true
        Equalizer.LowGain = 8
        Equalizer.MidGain = 0
        Equalizer.HighGain = 0
    elseif Settings.Enhancement == "Vocal Boost" then
        Equalizer.Enabled = true
        Equalizer.LowGain = -2
        Equalizer.MidGain = 6
        Equalizer.HighGain = 2
    elseif Settings.Enhancement == "Reverb" then
        Reverb.Enabled = true
        Reverb.DecayTime = 1.5
        Reverb.Density = 0.7
        Reverb.Diffusion = 0.7
    end
end

local function loadSong(index, play)
    State.SongIndex = math.clamp(index, 1, #Library)

    local song = Library[State.SongIndex]

    SongName.Text = song.Name
    Cover.Image = getcustomasset(song.Thumbnail)

    Sound:Stop()
    Sound.SoundId = getcustomasset(song.File)
    Sound.Volume = Settings.Volume
    Sound.PlaybackSpeed = Settings.PlaybackSpeed

    if play then
        Sound.TimePosition = Settings.UseStartingPosition and song.StartPosition or 0
        Sound:Play()
        State.Paused = false
        Pause.Text = "Ⅱ"
    end

    applyEnhancement()
end

local function previousSong()
    if #Library <= 1 then
        Sound.TimePosition = 0
        return
    end

    local index = State.SongIndex - 1

    if index < 1 then
        index = #Library
    end

    loadSong(index, true)
end

local function nextSong()
    if #Library <= 1 then
        Sound.TimePosition = 0
        return
    end

    local index = State.SongIndex + 1

    if index > #Library then
        index = 1
    end

    loadSong(index, true)
end

local function sameSong()
    Sound:Stop()
    Sound.TimePosition = 0
    Sound:Play()
end

local function randomSong()
    if #Library <= 1 then
        sameSong()
        return
    end

    local index

    repeat
        index = math.random(1, #Library)
    until index ~= State.SongIndex

    loadSong(index, true)
end

local function smoothPause()
    local info = TweenInfo.new(
        0.35,
        Enum.EasingStyle.Cubic,
        Enum.EasingDirection.Out
    )

    local tween = TweenService:Create(Sound, info, {
        Volume = 0,
        PlaybackSpeed = math.max(Settings.PlaybackSpeed * 0.15, 0.05)
    })

    tween:Play()

    tween.Completed:Once(function()
        Sound:Pause()
        Sound.Volume = Settings.Volume
        Sound.PlaybackSpeed = Settings.PlaybackSpeed
    end)
end

local function seek(input)
    local x = input.Position.X
    local start = ProgressBack.AbsolutePosition.X
    local width = ProgressBack.AbsoluteSize.X

    local percentage = math.clamp(
        (x - start) / width,
        0,
        1
    )

    if Sound.TimeLength > 0 then
        Sound.TimePosition = Sound.TimeLength * percentage
    end
end

ProgressBack.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        seek(input)
    end
end)

local dragging = false
local dragStart
local startPosition

Header.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    dragging = true
    dragStart = input.Position
    startPosition = Main.Position

    input.Changed:Connect(function()
        if input.UserInputState == Enum.UserInputState.End then
            dragging = false
        end
    end)
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart

    Main.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end)

RunService.RenderStepped:Connect(function()
    local length = Sound.TimeLength
    local position = Sound.TimePosition

    if length > 0 then
        Progress.Size = UDim2.fromScale(
            math.clamp(position / length, 0, 1),
            1
        )

        TimeLabel.Text =
            formatTime(position) ..
            " / " ..
            formatTime(length)
    end

    local loudness = math.clamp(
        Sound.PlaybackLoudness / 350,
        0,
        1
    )

    for i, bar in ipairs(Bars) do
        local centerDistance =
            math.abs(i - 16.5) / 16.5

        local strength =
            loudness * (1 - centerDistance * 0.45)

        local height =
            math.max(4, strength * 75)

        bar.Size = UDim2.new(
            0,
            4,
            0,
            height
        )
    end
end)
